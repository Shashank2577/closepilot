### Task P3-1: DB Migration for `deal_embeddings`

The `deal_embeddings` Drizzle schema exists in TypeScript but **no SQL migration has been generated** (journal only has entries 0000–0002). Without this, the table doesn't exist in the database and all vector search calls fail at runtime.

**Files:**
- Create: `packages/db/drizzle/0003_add_deal_embeddings.sql`
- Modify: `packages/db/drizzle/meta/_journal.json`

- [ ] **Step 1.1: Create the migration SQL file**

Create `packages/db/drizzle/0003_add_deal_embeddings.sql`:

```sql
-- Enable pgvector extension (idempotent)
CREATE EXTENSION IF NOT EXISTS vector;

-- Create deal_embeddings table
CREATE TABLE IF NOT EXISTS "deal_embeddings" (
  "id" uuid PRIMARY KEY DEFAULT gen_random_uuid() NOT NULL,
  "deal_id" integer NOT NULL REFERENCES "deals"("id") ON DELETE CASCADE,
  "embedding" vector(1536) NOT NULL,
  "content_type" text NOT NULL,
  "content" text NOT NULL,
  "created_at" timestamp NOT NULL DEFAULT now()
);

-- HNSW index for fast cosine-similarity search (pgvector >= 0.5)
CREATE INDEX IF NOT EXISTS "deal_embeddings_embedding_idx"
  ON "deal_embeddings"
  USING hnsw ("embedding" vector_cosine_ops)
  WITH (m = 16, ef_construction = 64);
```

import { UserRole } from '@closepilot/core';

export function hasRole(userRole: UserRole | undefined, allowedRoles: UserRole[]): boolean {
  if (!userRole) return false;
  return allowedRoles.includes(userRole);
}

export function canDeleteDeal(userRole: UserRole | undefined): boolean {
  return hasRole(userRole, [UserRole.ADMIN, UserRole.MANAGER]);
}

export function canEditDeal(_userRole: UserRole | undefined): boolean {
  return _userRole !== undefined;
}

```tsx
'use client';

import React, { createContext, useContext, useState, useEffect } from 'react';
import { UserRole } from '@closepilot/core';

interface UserRoleContextValue {
  role: UserRole;
  setRole: (role: UserRole) => void;
}

const UserRoleContext = createContext<UserRoleContextValue>({
  role: UserRole.ADMIN, // default to ADMIN in dev
  setRole: () => {},
});

export function UserRoleProvider({ children }: { children: React.ReactNode }) {
  const [role, setRoleState] = useState<UserRole>(UserRole.ADMIN);

  useEffect(() => {
    // Read persisted role from localStorage (set by login flow or dev override)
    const stored = localStorage.getItem('cp_user_role') as UserRole | null;
    if (stored && Object.values(UserRole).includes(stored)) {
      setRoleState(stored);
    }
  }, []);

  const setRole = (newRole: UserRole) => {
    localStorage.setItem('cp_user_role', newRole);
    setRoleState(newRole);
  };

  return (
    <UserRoleContext.Provider value={{ role, setRole }}>
      {children}
    </UserRoleContext.Provider>
  );
}

export function useUserRole(): UserRoleContextValue {
  return useContext(UserRoleContext);
}
```

```typescript
import { Hono } from 'hono';
import { requireRole } from '../middleware/auth.js';
import { getDb } from '@closepilot/db';
import { users } from '@closepilot/db';
import { eq } from 'drizzle-orm';
import { UserRole } from '@closepilot/core';
import { errorResponse } from '../lib/errors.js';
import type { AppContext } from '../types.js';

export const usersRoutes = new Hono<AppContext>();

usersRoutes.get('/', requireRole([UserRole.ADMIN]), async (c) => {
  const db = getDb();
  const allUsers = await db.select().from(users);
  return c.json({ users: allUsers });
});

usersRoutes.get('/:id', requireRole([UserRole.ADMIN, UserRole.MANAGER]), async (c) => {
  const id = parseInt(c.req.param('id'), 10);
  if (isNaN(id)) return c.json(errorResponse('Invalid user ID'), 400);
  const db = getDb();
  const [user] = await db.select().from(users).where(eq(users.id, id)).limit(1);
  if (!user) return c.json(errorResponse('User not found'), 404);
  return c.json(user);
});

usersRoutes.post('/', requireRole([UserRole.ADMIN]), async (c) => {
  const body = await c.req.json<{
    email: string;
    name: string;
    role: UserRole;
    orgId?: number;
  }>();
  if (!body.email || !body.name || !body.role) {
    return c.json(errorResponse('email, name, and role are required'), 400);
  }
  const db = getDb();
  const [created] = await db
    .insert(users)
    .values({ email: body.email, name: body.name, role: body.role, orgId: body.orgId })
    .returning();
  return c.json(created, 201);
});

usersRoutes.delete('/:id', requireRole([UserRole.ADMIN]), async (c) => {
  const id = parseInt(c.req.param('id'), 10);
  if (isNaN(id)) return c.json(errorResponse('Invalid user ID'), 400);
  const db = getDb();
  await db.delete(users).where(eq(users.id, id));
  return c.json({ success: true });
});
```

```typescript
import { DealStage } from '@closepilot/core';
import { getDb, deals, eq, or } from '@closepilot/db';
import { enqueueAgentJob } from './queue.js';
import type { AgentJob } from './jobs.js';

const CHASE_STAGES: DealStage[] = [
  DealStage.PROPOSAL_SENT,
  DealStage.FOLLOW_UP_1,
  DealStage.FOLLOW_UP_2,
  DealStage.NEGOTIATING,
  DealStage.STALLED,
];

export interface ChaseSchedulerDeps {
  enqueueJob?: (job: Pick<AgentJob, 'type' | 'dealId'>) => Promise<void>;
  getStalledDeals?: () => Promise<Array<{ id: string | number; stage: string }>>;
}

export class ChaseScheduler {
  private readonly enqueueJob: (job: Pick<AgentJob, 'type' | 'dealId'>) => Promise<void>;
  private readonly getStalledDeals: () => Promise<Array<{ id: string | number; stage: string }>>;

  constructor(deps: ChaseSchedulerDeps = {}) {
    this.enqueueJob =
      deps.enqueueJob ??
      ((job) => enqueueAgentJob({ type: job.type as AgentJob['type'], dealId: String(job.dealId) }));

    this.getStalledDeals =
      deps.getStalledDeals ??
      (async () => {
        const db = getDb();
        return db
          .select({ id: deals.id, stage: deals.stage })
          .from(deals)
          .where(or(...CHASE_STAGES.map((s) => eq(deals.stage, s))));
      });
  }

  async runSweep(): Promise<void> {
    const activeDeals = await this.getStalledDeals();
    console.log(`[ChaseScheduler] Sweeping ${activeDeals.length} active chase deals`);

    for (const deal of activeDeals) {
      await this.enqueueJob({ type: 'RunChase', dealId: String(deal.id) }).catch((err) =>
        console.error(`[ChaseScheduler] Failed to enqueue RunChase for deal ${deal.id}:`, err)
      );
    }
  }
}
```
